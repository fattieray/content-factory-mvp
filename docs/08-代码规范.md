## 三、命名规范

### 3.1 文件命名

| 类型 | 规范 | 示例 |
|------|------|------|
| React 组件 | PascalCase | `UserProfile.tsx` |
| Hook 文件 | camelCase + hook | `useAuth.ts` |
| 工具函数 | camelCase | `dateFormatter.ts` |
| 类型定义 | PascalCase + types | `userTypes.ts` |
| 常量文件 | UPPERCASE | `constants.ts` |
| Python 模块 | snake_case | `user_service.py` |
| API 路由 | kebab-case | `user-profile.py` |

### 3.2 变量命名

```typescript
// ✅ 正确
const userName: string;
const isLoggedIn: boolean;
const getUserInfo = () => {};
const MAX_RETRY_COUNT = 3;

// ❌ 错误
const username: string;  // 缺少语义
const flag: boolean;     // 无意义
const fn = () => {};     // 无意义
const max = 3;           // 缺少上下文
```

### 3.3 组件命名

```typescript
// ✅ 正确：组件名与文件名一致
export function UserProfile() {}

// ✅ 正确：带命名空间的组件
UserProfile.Avatar = function Avatar() {}
UserProfile.Info = function Info() {}

// ❌ 错误
export default function userProfile() {}  // 小写
```

## 四、代码规范

### 4.1 TypeScript 规范

```typescript
// ✅ 正确：明确的类型定义
interface User {
  id: string;
  name: string;
  email?: string;  // 可选字段用 ?
}

// ✅ 正确：使用类型别名处理联合类型
type Status = 'pending' | 'processing' | 'completed' | 'failed';

// ✅ 正确：泛型约束
function getData<T extends Record<string, unknown>>(data: T): T {
  return data;
}

// ❌ 错误：避免使用 any
const data: any;  // 使用 unknown 或具体类型

// ❌ 错误：避免隐式 any
function process(data) {}  // 应添加类型注解
```

### 4.2 React 规范

```typescript
// ✅ 正确：函数组件 + Hooks
function UserProfile({ userId }: UserProfileProps) {
  const { data, isLoading } = useUser(userId);
  
  if (isLoading) return <Loading />;
  if (!data) return <Error />;
  
  return <div>{data.name}</div>;
}

// ✅ 正确：使用 useMemo 优化
const filteredList = useMemo(() => {
  return list.filter(item => item.active);
}, [list]);

// ✅ 正确：事件处理命名
const handleSave = () => {};
const handleChange = (value: string) => {};

// ❌ 错误：避免使用 class 组件
class UserProfile extends React.Component {}

// ❌ 错误：避免在 render 中创建函数
<button onClick={() => handleSave()} />  // 应使用 useCallback
```

### 4.3 Python 规范

```python
# ✅ 正确：类型注解
def get_user(user_id: str) -> User | None:
    """获取用户信息"""
    pass

# ✅ 正确：使用 dataclass
from dataclasses import dataclass

@dataclass
class UserCreate:
    name: str
    email: str

# ✅ 正确：异常处理
try:
    result = process_data(data)
except ValidationError as e:
    logger.error(f"验证失败：{e}")
    raise HTTPException(status_code=400, detail=str(e))

# ❌ 错误：裸 except
try:
    process()
except:  # 应指定具体异常类型
    pass

# ❌ 错误：魔术数字
if status == 1:  # 应使用常量
    pass
```

## 七、Git 规范

### 7.1 Commit Message 格式

```
<type>(<scope>): <subject>

<body>

<footer>
```

### 7.2 Type 类型

| 类型 | 说明 |
|------|------|
| feat | 新功能 |
| fix | Bug 修复 |
| docs | 文档更新 |
| style | 代码格式（不影响功能） |
| refactor | 重构 |
| test | 测试相关 |
| chore | 构建/工具/配置 |

### 7.3 示例

```bash
# ✅ 正确
git commit -m "feat(auth): 添加用户登录功能"
git commit -m "fix(api): 修复生成接口超时问题"
git commit -m "docs: 更新 API 文档"

# ❌ 错误
git commit -m "更新代码"
git commit -m "fix bug"
```

### 7.4 分支命名

```
main              # 主分支
develop           # 开发分支
feature/xxx       # 功能分支
bugfix/xxx        # Bug 修复分支
release/v1.0.0    # 发布分支
hotfix/xxx        # 紧急修复分支
```

## 八、安全规范

### 8.1 前端安全

```typescript
// ✅ 正确：XSS 防护
// 使用 React 默认转义，避免 dangerouslySetInnerHTML
<div>{userInput}</div>  // 安全

// ✅ 正确：敏感信息处理
// 不将 token 存储在 localStorage
// 使用 httpOnly cookie

// ❌ 错误
<div dangerouslySetInnerHTML={{ __html: userInput }} />  // XSS 风险
```

### 8.2 后端安全

```python
# ✅ 正确：参数验证
from pydantic import BaseModel, EmailStr

class UserCreate(BaseModel):
    email: EmailStr  # 自动验证邮箱格式
    password: str = Field(min_length=8)  # 密码长度验证

# ✅ 正确：SQL 注入防护
# 使用 ORM 参数化查询
user = await db.query(User).filter(User.email == email).first()

# ✅ 正确：敏感信息加密
from passlib.context import CryptContext

pwd_context = CryptContext(schemes=["bcrypt"])
hashed = pwd_context.hash(password)

# ❌ 错误
query = f"SELECT * FROM users WHERE email = '{email}'"  # SQL 注入风险
```
