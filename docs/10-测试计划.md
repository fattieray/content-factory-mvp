```typescript
// __tests__/utils/dateFormatter.test.ts
import { formatDate, formatDateRelative } from '@/utils/dateFormatter';

describe('dateFormatter', () => {
  describe('formatDate', () => {
    it('应该格式化日期为 YYYY-MM-DD', () => {
      const date = new Date('2024-01-15T10:30:00Z');
      expect(formatDate(date)).toBe('2024-01-15');
    });
    
    it('应该处理无效日期', () => {
      expect(formatDate(null)).toBe('');
      expect(formatDate(undefined)).toBe('');
    });
  });
  
  describe('formatDateRelative', () => {
    it('应该显示"刚刚"', () => {
      const now = new Date();
      expect(formatDateRelative(now)).toBe('刚刚');
    });
    
    it('应该显示"X 小时前"', () => {
      const threeHoursAgo = new Date(Date.now() - 3 * 60 * 60 * 1000);
      expect(formatDateRelative(threeHoursAgo)).toBe('3 小时前');
    });
  });
});
```

```python
import pytest
from app.services.style_analyzer import StyleAnalyzer

class TestStyleAnalyzer:
    """风格分析器测试"""
    
    def setup_method(self):
        self.analyzer = StyleAnalyzer()
    
    def test_extract_surface_features(self):
        """测试表面特征提取"""

        """测试关键词提取"""
        text = "人工智能是未来技术发展的重要方向，AI 将改变世界"
        keywords = self.analyzer.extract_keywords(text, top_k=5)
        
        assert len(keywords) <= 5

    def test_invalid_input(self):
        """测试无效输入处理"""
        with pytest.raises(ValueError):
            self.analyzer.analyze("")
        
        with pytest.raises(ValueError):
            self.analyzer.analyze(None)
```

### 3.1 API 集成测试

**测试覆盖**：
- 所有 REST API 端点
- 请求验证
- 响应格式
- 错误处理

**示例**：

```python
# tests/api/test_generations.py
import pytest
from fastapi.testclient import TestClient
from app.main import app

client = TestClient(app)

class TestGenerationsAPI:
    """生成接口测试"""
    
    @pytest.fixture
    def auth_headers(self, test_user):
        """认证头"""
        token = get_auth_token(test_user)
        return {"Authorization": f"Bearer {token}"}
    
    def test_create_generation(self, auth_headers):
        """测试创建生成任务"""
        response = client.post(
            "/api/v1/generations",
            headers=auth_headers,
            json={
                "topics": ["AI 发展趋势", "创业心得"],
                "style_model_id": "test-model-id"
            }
        )
        
        assert response.status_code == 201
        data = response.json()
        assert data["success"] is True
        assert "id" in data["data"]
        assert data["data"]["status"] == "pending"
    
    def test_create_generation_invalid_input(self, auth_headers):
        """测试无效输入"""
        response = client.post(
            "/api/v1/generations",
            headers=auth_headers,
            json={"topics": []}  # 空主题
        )
        
        assert response.status_code == 400
        assert "error" in response.json()
    
    def test_get_generation(self, auth_headers, test_generation):
        """测试获取生成任务"""
        response = client.get(
            f"/api/v1/generations/{test_generation.id}",
            headers=auth_headers
        )
        
        assert response.status_code == 200
        assert response.json()["data"]["id"] == str(test_generation.id)
    
    def test_unauthorized_access(self):
        """测试未授权访问"""
        response = client.get("/api/v1/generations/123")
        assert response.status_code == 401
```

```typescript
// __tests__/pages/StyleTraining.test.tsx
import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import { StyleTraining } from '@/pages/StyleTraining';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';

const renderWithProviders = (component: ReactElement) => {
  const queryClient = new QueryClient({
    defaultOptions: {
      queries: { retry: false },
    },
  });
  
  return render(
    <QueryClientProvider client={queryClient}>
      {component}
    </QueryClientProvider>
  );
};

describe('StyleTraining', () => {
  it('应该显示上传区域', () => {
    renderWithProviders(<StyleTraining />);
    
    expect(screen.getByText(/上传样本文章/i)).toBeInTheDocument();
  });
  
  it('应该允许上传文章', async () => {
    renderWithProviders(<StyleTraining />);
    
    const fileInput = screen.getByTestId('file-input');
    const file = new File(['测试内容'], 'test.md', { type: 'text/markdown' });
    
    fireEvent.change(fileInput, { target: { files: [file] } });
    
    await waitFor(() => {
      expect(screen.getByText('test.md')).toBeInTheDocument();
    });
  });
  
  it('应该显示训练进度', async () => {
    renderWithProviders(<StyleTraining />);
    
    // 模拟开始训练
    fireEvent.click(screen.getByText('开始训练'));
    
    await waitFor(() => {
      expect(screen.getByText(/训练中/i)).toBeInTheDocument();
      expect(screen.getByRole('progressbar')).toBeInTheDocument();
    });
  });
});
```

```typescript
// e2e/tests/content-generation.spec.ts
import { test, expect } from '@playwright/test';

test.describe('内容生成流程', () => {
  test.beforeEach(async ({ page }) => {
    // 登录
    await page.goto('/login');
    await page.fill('[name="email"]', 'test@example.com');
    await page.fill('[name="password"]', 'password123');
    await page.click('button[type="submit"]');
    await expect(page).toHaveURL('/dashboard');
  });
  
  test('E2E-003: 完整的内容生成流程', async ({ page }) => {
    // 1. 进入生成页面
    await page.click('[data-testid="create-generation"]');
    await expect(page).toHaveURL('/generate');
    
    // 2. 选择风格模型
    await page.click('[data-testid="style-model-select"]');
    await page.click('[data-testid="model-option-1"]');
    
    // 3. 输入主题
    await page.fill('[name="topic-1"]', 'AI 产品经理的核心能力');
    await page.fill('[name="points-1"]', '技术理解力，用户洞察，商业敏感度');
    
    // 4. 开始生成
    await page.click('[data-testid="start-generation"]');
    
    // 5. 等待生成完成
    await expect(page.locator('[data-testid="progress"]')).toBeVisible();
    await expect(page.locator('[data-testid="generation-complete"]'))
      .toBeVisible({ timeout: 60000 });
    
    // 6. 验证生成结果
    const versions = page.locator('[data-testid="version-card"]');
    await expect(versions).toHaveCount(4);
    
    // 7. 选择一个版本
    await page.click('[data-testid="version-card-2"] [data-testid="select"]');
    await expect(page.locator('[data-testid="selected-badge"]'))
      .toBeVisible();
    
    // 8. 导出内容
    await page.click('[data-testid="export"]');
    await page.click('[data-testid="export-markdown"]');
    
    // 验证下载
    const download = await page.waitForEvent('download');
    expect(download.suggestedFilename()).toContain('.md');
  });
});
```

```javascript
// k6/load-test.js
import http from 'k6/http';
import { check, sleep } from 'k6';

export const options = {
  stages: [
    { duration: '2m', target: 10 },   // 热身到 10 用户
    { duration: '5m', target: 50 },   // 增加到 50 用户
    { duration: '10m', target: 50 },  // 保持 50 用户
    { duration: '5m', target: 100 },  // 增加到 100 用户
    { duration: '10m', target: 100 }, // 保持 100 用户
    { duration: '5m', target: 0 },    // 逐渐降低到 0
  ],
  thresholds: {
    http_req_duration: ['p(95)<500'], // P95 < 500ms
    http_req_failed: ['rate<0.01'],   // 错误率 < 1%
  },
};

export default function () {
  const token = __ENV.AUTH_TOKEN;
  const headers = { Authorization: `Bearer ${token}` };
  
  // 获取生成列表
  const listRes = http.get('https://api.example.com/v1/generations', {
    headers,
  });
  check(listRes, { 'list status is 200': (r) => r.status === 200 });
  
  sleep(1);
  
  // 创建生成任务
  const createRes = http.post(
    'https://api.example.com/v1/generations',
    JSON.stringify({
      topics: ['测试主题'],
      style_model_id: 'test-model',
    }),
    { headers }
  );
  check(createRes, { 'create status is 201': (r) => r.status === 201 });
  
  sleep(2);
}
```
